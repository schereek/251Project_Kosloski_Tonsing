table_data <- ca_id_rate %>%
  group_by(race_ethnicity, sex) %>%
  summarise(rate = (sum(new_infections) / sum(population)) * 100000,
            .groups = "drop") %>%
  tidyr::pivot_wider(names_from = sex, values_from = rate) %>%
  mutate(
    Overall = (Female + Male) / 2,
    race_ethnicity = fct_reorder(race_ethnicity, Overall, .desc = TRUE)) %>%
  arrange(race_ethnicity) %>%
  mutate(
    Female = comma(round(Female, 1)),
    Male = comma(round(Male, 1)),
    Overall = comma(round(Overall, 1)))

age_plot<-ca_id_rate%>%
  group_by(age_cat, county) %>%                
  summarise(rate = sum(new_infections)/sum(population)*100000)

plot_ly(
  age_plot,
  x = ~age_cat,
  y = ~rate,
  type = "box",
  color = ~age_cat,
  hoverinfo = "y+text",
  text = ~paste("County:", county)
)%>%
  layout(
    title = "Novel Respiratory Disease in California: Rates by Age Group and County",
    yaxis = list(title = "Rate per 100,000"),
    xaxis = list(title = "Age Group"),
    boxmode = "group",
    paper_bgcolor = "azure",
    plot_bgcolor = "white"
  )

Interpretation: While median values for infection rates vary between age groups, the spread within the age groups is quite wide and not normally distributed even when agregated at the county level. This suggests that individual level variations (perhaps due to underlying health conditions, behavioral patterns, etc.) may be more significant than age alone in influencing disease risk.
